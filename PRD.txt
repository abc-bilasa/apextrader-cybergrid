1. Executive Summary & Objective
ApexTrader Cyber-Grid is an advanced 42-level trading account scaling, risk management, and journal application formatted with a high-tech dark cyberpunk/neon aesthetic.
The app manages capital progression starting from ₹10,000 (INR) through 42 levels using dynamic compounding math. It automatically enforces trading risk rules:
Promotion: Achieving a +50% profit target relative to the level's baseline capital promotes the account by +1 level (up to Level 42).
Demotion Penalty: Incurring a -30% daily loss (or -60% maximum loss) demotes the account by -2 levels (down to Level 1 minimum).
All calculations, progression logic, state checks, and audit trails are computed server-side inside Supabase using an atomic PostgreSQL Stored Procedure (RPC) to eliminate client-side tampering.
2. Complete Mathematical Model
The application uses dynamic mathematical compounding rather than hardcoded tables.
2.1 Baseline Formulas (Level N)
For any given Level N (where 1≤N≤42):
Baseline Capital (CapN):
CapN=10,000×(1.5)N-1
(Level 1 = ₹10,000 | Level 2 = ₹15,000 | Level 3 = ₹22,500 | Level 4 = ₹33,750 ...)
Daily Drawdown Limit (DDdaily):$$DD_{\text{daily}} = 30\% \times \text{Cap}_N = 0.30 \times \text{Cap}_N$$
Maximum Allowable Drawdown Limit (DDmax):$$DD_{\text{max}} = 60\% \times \text{Cap}_N = 0.60 \times \text{Cap}_N$$
Level Target Profit (Tplan):$$T_{\text{plan}} = 50\% \times \text{Cap}_N = 0.50 \times \text{Cap}_N$$
Daily Milestones (D1, D2, D3, D4 Breakdown):
D1 (10%): 0.10×CapN
D2 (10%): 0.10×CapN
D3 (10%): 0.10×CapN
D4 (20%): 0.20×CapN
Total: $\text{D1} + \text{D2} + \text{D3} + \text{D4} = 50\%$
Target Final Capital (Capfinal):
Capfinal=CapN+Tplan=1.50×CapN

EXTENSIONS-- ============================================================================CREATE EXTENSION IF NOT EXISTS "uuid-ossp";-- ============================================================================-- 2. TABLES DEFINITION-- ============================================================================-- Table A: Trading Accounts (One record per user)CREATE TABLE public.trading_accounts (  id UUID DEFAULT uuid_generate_v4() PRIMARY KEY,  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE NOT NULL UNIQUE,  current_level INT DEFAULT 1 NOT NULL CHECK (current_level BETWEEN 1 AND 42),  current_capital NUMERIC(14, 2) DEFAULT 10000.00 NOT NULL,  peak_capital NUMERIC(14, 2) DEFAULT 10000.00 NOT NULL,  created_at TIMESTAMPTZ DEFAULT NOW() NOT NULL,  updated_at TIMESTAMPTZ DEFAULT NOW() NOT NULL);-- Table B: Trades / Journal EntriesCREATE TABLE public.trades (  id UUID DEFAULT uuid_generate_v4() PRIMARY KEY,  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE NOT NULL,  level INT NOT NULL CHECK (level BETWEEN 1 AND 42),  starting_capital NUMERIC(14, 2) NOT NULL,  pnl NUMERIC(14, 2) NOT NULL,  ending_capital NUMERIC(14, 2) NOT NULL,  trade_type TEXT CHECK (trade_type IN ('PROFIT', 'LOSS', 'NEUTRAL')) NOT NULL,  notes TEXT,  created_at TIMESTAMPTZ DEFAULT NOW() NOT NULL);-- Table C: Audit Log for Level ChangesCREATE TABLE public.level_history (  id UUID DEFAULT uuid_generate_v4() PRIMARY KEY,  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE NOT NULL,  old_level INT NOT NULL CHECK (old_level BETWEEN 1 AND 42),  new_level INT NOT NULL CHECK (new_level BETWEEN 1 AND 42),  trigger_reason TEXT CHECK (trigger_reason IN ('TARGET_MET', 'DRAWDOWN_HIT', 'MANUAL_RESET')) NOT NULL,  capital_at_transition NUMERIC(14, 2) NOT NULL,  created_at TIMESTAMPTZ DEFAULT NOW() NOT NULL);-- ============================================================================-- 3. INDEXES FOR PERFORMANCE-- ============================================================================CREATE INDEX idx_trades_user_created ON public.trades(user_id, created_at ASC);CREATE INDEX idx_level_history_user_created ON public.level_history(user_id, created_at DESC);-- ============================================================================-- 4. ROW LEVEL SECURITY (RLS)-- ============================================================================ALTER TABLE public.trading_accounts ENABLE ROW LEVEL SECURITY;ALTER TABLE public.trades ENABLE ROW LEVEL SECURITY;ALTER TABLE public.level_history ENABLE ROW LEVEL SECURITY;-- Policies for trading_accountsCREATE POLICY "Users can view own account"   ON public.trading_accounts FOR SELECT USING (auth.uid() = user_id);CREATE POLICY "Users can update own account"   ON public.trading_accounts FOR UPDATE USING (auth.uid() = user_id);-- Policies for tradesCREATE POLICY "Users can view own trades"   ON public.trades FOR SELECT USING (auth.uid() = user_id);CREATE POLICY "Users can insert own trades"   ON public.trades FOR INSERT WITH CHECK (auth.uid() = user_id);-- Policies for level_historyCREATE POLICY "Users can view own level history"   ON public.level_history FOR SELECT USING (auth.uid() = user_id);CREATE POLICY "Users can insert own level history"   ON public.level_history FOR INSERT WITH CHECK (auth.uid() = user_id);-- ============================================================================-- 5. AUTOMATIC ACCOUNT CREATION ON USER SIGNUP-- ============================================================================CREATE OR REPLACE FUNCTION public.handle_new_user()RETURNS TRIGGER AS $$BEGIN  INSERT INTO public.trading_accounts (user_id, current_level, current_capital, peak_capital)  VALUES (NEW.id, 1, 10000.00, 10000.00);  RETURN NEW;END;$$ LANGUAGE plpgsql SECURITY DEFINER;CREATE TRIGGER on_auth_user_created  AFTER INSERT ON auth.users  FOR EACH ROW EXECUTE FUNCTION public.handle_new_user();-- ============================================================================-- 6. ATOMIC PROGRESSION & RISK ENGINE (RPC FUNCTION)-- ============================================================================CREATE OR REPLACE FUNCTION public.log_trade(  p_pnl NUMERIC(14, 2),  p_notes TEXT DEFAULT NULL)RETURNS JSONB AS $$DECLARE  v_user_id UUID := auth.uid();  v_acc RECORD;  v_base_cap NUMERIC(14, 2);  v_target NUMERIC(14, 2);  v_dd_limit NUMERIC(14, 2);  v_new_cap NUMERIC(14, 2);  v_new_level INT;  v_trigger TEXT := 'NONE';  v_trade_type TEXT;  v_old_level INT;BEGIN  -- Security check  IF v_user_id IS NULL THEN    RAISE EXCEPTION 'Not authenticated';  END IF;  -- Row lock account record  SELECT * INTO v_acc   FROM public.trading_accounts   WHERE user_id = v_user_id   FOR UPDATE;  IF NOT FOUND THEN    RAISE EXCEPTION 'Account not found';  END IF;  v_old_level := v_acc.current_level;  -- Baseline formula calculations for level  v_base_cap := 10000.00 * POWER(1.5, v_old_level - 1);  v_target := v_base_cap * 0.50;      -- 50% target  v_dd_limit := v_base_cap * 0.30;    -- 30% daily loss limit  -- Calculate updated capital  v_new_cap := GREATEST(0.00, v_acc.current_capital + p_pnl);  v_new_level := v_old_level;  -- Determine Trade Type  IF p_pnl > 0 THEN v_trade_type := 'PROFIT';  ELSIF p_pnl < 0 THEN v_trade_type := 'LOSS';  ELSE v_trade_type := 'NEUTRAL';  END IF;  -- Rule Execution:  -- Target Hit (+50% profit above level baseline)  IF (v_new_cap - v_base_cap) >= v_target AND v_old_level < 42 THEN    v_new_level := LEAST(42, v_old_level + 1);    v_trigger := 'TARGET_MET';  -- Drawdown Limit Hit (-30% loss from level baseline)  ELSIF (v_base_cap - v_new_cap) >= v_dd_limit THEN    v_new_level := GREATEST(1, v_old_level - 2);    v_trigger := 'DRAWDOWN_HIT';  END IF;  -- Update Trading Account  UPDATE public.trading_accounts  SET     current_capital = v_new_cap,    current_level = v_new_level,    peak_capital = GREATEST(peak_capital, v_new_cap),    updated_at = NOW()  WHERE user_id = v_user_id;  -- Insert Trade Journal Row  INSERT INTO public.trades (user_id, level, starting_capital, pnl, ending_capital, trade_type, notes)  VALUES (v_user_id, v_old_level, v_acc.current_capital, p_pnl, v_new_cap, v_trade_type, p_notes);  -- Insert Level History Audit Row if level changed  IF v_trigger != 'NONE' THEN    INSERT INTO public.level_history (user_id, old_level, new_level, trigger_reason, capital_at_transition)    VALUES (v_user_id, v_old_level, v_new_level, v_trigger, v_new_cap);  END IF;  RETURN jsonb_build_object(    'success', true,    'old_level', v_old_level,    'new_level', v_new_level,    'new_capital', v_new_cap,    'trigger', v_trigger,    'level_changed', (v_trigger != 'NONE')  );END;$$ LANGUAGE plpgsql SECURITY DEFINER;
